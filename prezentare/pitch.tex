\section{Solution}

% ================================================================== 4 · architecture: algorithms + rules
\begin{frame}{Architecture: computer vision first, ML only where it pays}
  \begin{columns}[T]
    \begin{column}{0.64\textwidth}
      \sxstep{1}{Row axes: a 1-D signal.}{Dominant angle, projection on the row normal, autocorrelation gives the spacing; SNR $\geq$ 3 rejects orchards; Huber line fit; union-find links rows across tiles.}
      \sxstep{2}{Canopies: one polygon per plant.}{Lab a* vegetation threshold, kept only inside the row corridor; connected components. A U-Net runs only where CV is blind (old vines in grass: 88 tiles, 13\,s).}
      \sxstep{3}{Attributes: explicit rules.}{Gap $\geq$ 5\,m $\Rightarrow$ row \emph{disrupted}; inter-row cover from the vegetation fraction; unclear image $\Rightarrow$ \emph{unassessable}.}
      \sxstep{4}{Waste: a cascade.}{Colour/shape rules $\rightarrow$ OpenCLIP probe $\rightarrow$ SAM 3 on the top crops $\rightarrow$ a human approves every box.}
      \sxstep{5}{Blocks, farms, route.}{Cadastre parcels as a prior, road cuts; walking graph on inter-rows + passages; generalised TSP (OR-Tools, 2-opt), closed at START, $\leq$ 2\,\% outside.}
    \end{column}
    \begin{column}{0.32\textwidth}
      \sxstat{$\approx$\,3 min}{CV on all 311 tiles, laptop CPU (measured)}
      \vspace{2mm}

      \sxstat{0 labels}{no training set needed; thresholds set on the 2 example tiles}
      \vspace{2mm}

      \sxstat{$\approx$\,18 h}{SAM 3 on every tile instead (extrapolated)}
      \vspace{2mm}

      {\tiny\color{sxMuted} On the 2 labelled tiles CV scores 0.816 vs 0.641 for the U-Net alone (canopy).\par}
    \end{column}
  \end{columns}
\end{frame}

    \foreach \a/\b in {in/per, per/mc, mc/post, post/out} \draw[arr] (\a) -- (\b);
    \node[below=1mm of mc, font=\tiny, text=sxAmber] {human in the loop};
  \end{tikzpicture}

  \vspace{6mm}
  \begin{columns}[T]
    \begin{column}{0.235\textwidth}\begin{sxcard}{$\approx$ 17 min}Tiles to deliverables on one laptop (M3 Pro), CPU only, 8 workers.\end{sxcard}\end{column}
    \begin{column}{0.235\textwidth}\begin{sxcard}{Cached by config hash}Re-runs reuse per-tile caches; a new config never reads stale ones.\end{sxcard}\end{column}
    \begin{column}{0.235\textwidth}\begin{sxcard}{One source of truth}Every threshold lives in \texttt{default.yaml}; every run logs its config and timings.\end{sxcard}\end{column}
    \begin{column}{0.235\textwidth}\begin{sxcard}{Reproducible}Pinned \texttt{uv.lock}, 2\,378 tests, CI, Docker with \texttt{-{}-network none}. No paid API, no LLM at runtime.\end{sxcard}\end{column}
  \end{columns}
\end{frame}

      \begin{tabular}{@{}p{0.70\linewidth} r@{}}
        \toprule
        \textbf{One-time setup} & \textbf{hours}\\

\begin{frame}{Scaling to industry}
  \begin{columns}[T]
    \begin{column}{0.30\textwidth}

      \vspace{2mm}

      \sxstat{$\approx$\,130 h}{all $\approx$\,100\,000\,ha of Moldovan vineyards on one laptop; $\approx$\,16\,h on a 64-core server}
      \vspace{1mm}

      {\tiny\color{sxGrey} Linear extrapolation of a measured run. Tiles are independent, so it parallelises.\par}
    \end{column}
    \begin{column}{0.66\textwidth}
      {\scriptsize\bfseries\color{sxNight} Features we want to offer the industry\par}
      \vspace{1.5mm}
      \sxstep{1}{Ground robot, Street View for every row.}{It drives our route (14\,km instead of $\geq$\,41\,km), side cameras image two rows per pass, on-board ML flags disease symptoms, each image tied to its \texttt{row\_id}.}
      \sxstep{2}{Season-to-season change.}{Fly again, compare: new gaps, replanting, abandoned plots, per row.}
      \sxstep{3}{Registry and subsidy integration.}{Export per parcel to the vine registry (ONVV) and to AIPA field checks; cadastre (AGCC) parcels already on the map.}
      \sxstep{4}{Reports and routes per farm.}{Per-farm route and GPX for phones and robots; CSV reports by block and row.}
      \sxstep{5}{On government infrastructure.}{Offline Docker image, no external API; multi-user web app in RO / EN / RU. Other row crops (orchards, berries) reuse the same 1-D row model.}
    \end{column}
  \end{columns}
\end{frame}
